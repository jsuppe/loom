/**
 * src/aggregate.js
 * 
 * Groups events by a specified field and applies reducers to each group.
 * 
 * @param {Array<Object>} events - Array of event objects.
 * @param {Object} opts - Options object.
 * @param {string} opts.groupBy - Field name to group events by.
 * @param {Object} opts.reducers - Map of reducer keys to { initial, step } objects.
 * @returns {Object} - Aggregated result object.
 */

export function aggregate(events, opts) {
  const { groupBy, reducers } = opts;

  // Initialize the result accumulator
  const result = {};

  // Helper to initialize a group if it doesn't exist
  const ensureGroup = (key) => {
    if (!result[key]) {
      result[key] = {};
      // Initialize all reducers with their initial values
      for (const [reducerKey, { initial }] of Object.entries(reducers)) {
        result[key][reducerKey] = initial;
      }
    }
  };

  // Helper to apply a reducer step
  const applyReducer = (group, reducerKey, value) => {
    const current = group[reducerKey];
    const step = reducers[reducerKey].step;
    // Apply the step function. If step is undefined, return current.
    if (step) {
      group[reducerKey] = step(current, value);
    }
  };

  // Process each event
  for (const event of events) {
    const key = event[groupBy];
    ensureGroup(key);

    // Apply all reducers for this event
    for (const reducerKey of Object.keys(reducers)) {
      applyReducer(result[key], reducerKey, event);
    }
  }

  return result;
}
