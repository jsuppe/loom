/**
 * Validates an input object against a list of declarative rules.
 *
 * @param {Object} input - The object to validate.
 * @param {Array<{field: string, check: Function, message: string}>} rules - The validation rules.
 * @returns {{valid: boolean, errors: Array<{field: string, message: string}>}} The validation result.
 */
export function validate(input, rules) {
  const errors = [];

  for (const rule of rules) {
    const { field, check, message } = rule;
    const value = input[field];

    if (!check(value)) {
      errors.push({ field, message });
    }
  }

  return {
    valid: errors.length === 0,
    errors,
  };
}
