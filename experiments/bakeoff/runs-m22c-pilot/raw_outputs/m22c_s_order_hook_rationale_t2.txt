      if (!store.inventory.containsKey(sku)) {
        throw ArgumentError('SKU $sku not found in inventory');
      }
    }

    // 2. Body: Reserve stock for each line item
    // We need to track what we've reserved to release on failure.
    final reservedSkus = <String>[];
    final reservedQuantities = <String, int>{};

    for (final line in lines) {
      final sku = line.sku;
      final quantity = line.quantity;
      final inventoryItem = store.inventory[sku];

      if (inventoryItem == null) {
        // This should have been caught in pre-check, but safety first.
        throw ArgumentError('SKU $sku not found in inventory');
      }

      if (inventoryItem.availableQuantity < quantity) {
        throw ArgumentError('Insufficient stock for SKU $sku. Required: $quantity, Available: ${inventoryItem.availableQuantity}');
      }

      // Reserve stock
      await inventoryService.reserve(sku, quantity);
      reservedSkus.add(sku);
      reservedQuantities[sku] = (reservedQuantities[sku] ?? 0) + quantity;
    }

    // 3. Create the order with initial status 'placed'
    final order = Order(
      id: store.nextOrderId(),
      customerId: customerId,
      status: OrderStatus.placed,
      lines: lines,
      createdAt: DateTime.now(),
    );

    // 4. Persist the order
    store.orders.add(order);

    return order;
  }

  /// Marks an order as paid.
  Future<void> markPaid(String orderId) async {
    final order = store.orders.firstWhere(
      (o) => o.id == orderId,
      orElse: () => throw ArgumentError('Order $orderId not found'),
    );

    if (order.status != OrderStatus.placed) {
      throw ArgumentError('Cannot mark order $orderId as paid. Current status: ${order.status}');
    }

    order.status = OrderStatus.paid;
    order.updatedAt = DateTime.now();
  }
